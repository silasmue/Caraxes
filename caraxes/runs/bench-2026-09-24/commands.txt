## v1 backend from main (target/release/vhost-user-nvrm was a 2026-09-19 build, before the SHMEM protocol)
CARGO_TARGET_DIR=target/bench cargo build --release -p vhost-user-nvrm        # $B/build-vhost-user-nvrm.log

## v2 on vm1: refused while vm2's device holds its reservation (fixed address at the top of the host FB)
VM=vm1 CARAXES_IP=192.168.101.21 CARAXES_TAP=caraxes1 CARAXES_MAC=02:00:00:00:00:21 CARAXES_BOOT_MODEL=1 \
  CARAXES_HOST_RM=1 CARAXES_HOST_EXEC=1 CARAXES_MIRROR_CHANNELS=1 CARAXES_MIRROR_GR_SHADOW=1 \
  CARAXES_RESERVE_VRAM=1 CARAXES_DMA=register CARAXES_FB_MIB=1024 tools/caraxes-vm.sh up   # $B/v2-up1/run.txt

## v1 VM (overlay on the baked 610.57.04 image, seed of the v1check VM, tap caraxes5)
caraxes/tools/bench-v1vm.sh prepare && caraxes/tools/bench-v1vm.sh up && caraxes/tools/bench-v1vm.sh modules
CARAXES_IP=192.168.105.25 VM=bench-v1 tools/caraxes-vm.sh sync-tests      # Leandro-Test probes into the guest, built there

## One set of binaries: bench-cu built in the guest, the probes from sync-tests, copied to target/bench/bin
caraxes/tools/bench-v1vm.sh ssh 'cat > ~/bench/bench-cu.c' < caraxes/tools/bench-cu.c
caraxes/tools/bench-v1vm.sh ssh 'cd ~/bench && gcc -O2 -Wall -o bench-cu bench-cu.c -I<venv nvidia/cu13/include> -L/opt/nvrm/lib -lcuda'
# (then cat each of bench-cu nvprobe ioctlping ctrlping vrampress kernels.ptx back to target/bench/bin; sha256 in $B/*/series.txt)
caraxes/tools/bench-run.sh stage v1

## Series 1: native + v1 interleaved, N=5 (shows the v1 pinned-page retention from rep 2 on)
caraxes/tools/bench-run.sh series --targets "native v1" --reps 5 --out $B/series1
## v1 restarted, startup metric, series 1b with a v1 restart before every rep
caraxes/tools/bench-v1vm.sh down && caraxes/tools/bench-v1vm.sh up && caraxes/tools/bench-v1vm.sh modules
caraxes/tools/bench-run.sh startup v1 --out $B/series1b
V1_RESTART=1 caraxes/tools/bench-run.sh series --targets "native v1" --reps 5 --out $B/series1b
caraxes/tools/bench-v1vm.sh down        # host memory: swap full, RAM 25/31 GiB
python3 caraxes/tools/bench-summary.py $B/series1b/native $B/series1b/v1 > $B/summary-series1b.md

## v2 on vm1 after "vm2 free" (main 223075d, target/release/vfio-user-nvgpu sha256 d140e7cb...)
source $B/v2.env                       # VM=vm1, green configuration, CARAXES_FB_MIB=4096
tools/caraxes-vm.sh up && tools/caraxes-vm.sh wait 120 && tools/caraxes-vm.sh probe bench-load1     # $B/v2-up2/run.txt
tools/caraxes-vm.sh sync-tests && caraxes/tools/bench-run.sh stage v2
tools/caraxes-vm.sh ssh 'bash -s' < $B/v2-nodes.sh          # nvidia-uvm + /dev/nvidia* (no nvidia-modprobe in the image)
# context creation ~68 s (NV_UFLUSH timeouts): $B/v2-hang/sequence1.txt (after FLR + reload), then a fresh device:
tools/caraxes-vm.sh down; tools/caraxes-vm.sh up; tools/caraxes-vm.sh wait 120; tools/caraxes-vm.sh probe bench-load3   # $B/v2-up3
# 8 x nvprobe 1, timed: $B/v2-hang/sequence2-fresh.txt; one under strace + guest dmesg: $B/v2-hang/nvprobe1-*.txt
caraxes/tools/bench-run.sh startup v2 --out $B/series2
V2_RESTART=1 caraxes/tools/bench-run.sh series --targets "native v2" --reps 5 --out $B/series2

## Split of series2 by device build (04:20): rep 2 moved to $B/series2-rep2-build-c09fcccc (README there)
## Native-only reference on the quiet host
caraxes/tools/bench-run.sh series --targets native --reps 5 --out $B/series3-native-quiet

## Final series ("quiet host"), main 8d44ee6, device sha256 249c15ca...
cargo build --release -p vfio-user-nvgpu
source $B/v2.env; tools/caraxes-vm.sh up && tools/caraxes-vm.sh wait 120 && tools/caraxes-vm.sh probe bench-final && caraxes/tools/bench-run.sh v2-nodes   # $B/final/v2-up.txt
caraxes/tools/bench-v1vm.sh up && caraxes/tools/bench-v1vm.sh modules                       # $B/final/v1-up.txt
caraxes/tools/bench-run.sh startup v1 --out $B/final; caraxes/tools/bench-run.sh startup v2 --out $B/final
V1_RESTART=1 V2_RESTART=1 caraxes/tools/bench-run.sh series --targets "native v1 v2" --reps 5 --out $B/final
python3 caraxes/tools/bench-summary.py $B/final/v1 $B/final/v2 $B/final/native > $B/summary-final.md
